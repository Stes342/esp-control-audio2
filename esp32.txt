bool multiUserAuth(String &username) {
  if (!server.hasHeader("Authorization")) return false;

  String auth = server.header("Authorization");
  if (!auth.startsWith("Basic ")) return false;

  String encoded = auth.substring(6); // удалить "Basic "

  // Декодируем base64
  size_t len = encoded.length();
  size_t outLen = 0;
  unsigned char decoded[129]; // +1 для завершающего нуля
  int res = mbedtls_base64_decode(decoded, sizeof(decoded), &outLen,
                                  (const unsigned char *)encoded.c_str(), len);
  if (res != 0) return false;
  if (outLen >= sizeof(decoded)) return false;
  decoded[outLen] = '\0';

  String decodedStr = String((char *)decoded).substring(0, outLen);

  int sep = decodedStr.indexOf(':');
  if (sep == -1) return false;

  String user = decodedStr.substring(0, sep);
  String pass = decodedStr.substring(sep + 1);
  user.trim();
  pass.trim();
  Serial.println("String user_multiUserAuth:"  + user);
  Serial.println("String pass_multiUserAuth:"  + pass);

  for (int i = 1; i <= 4; i++) {
    String expectedUser = "user" + String(i);
    Serial.println("expectedUser:"  + expectedUser);
    if (user == expectedUser) {
      preferences.begin("users", true);
      String storedPass = preferences.getString((expectedUser + "_pass").c_str(), "");
      Serial.println("storedPass_multiUserAuth:"  + storedPass);
      preferences.end();
      if (storedPass != "" && pass == storedPass) {
        username = expectedUser;
        Serial.println("username_multiUserAuth:"  + username);
        Serial.println("expectedUser_multiUserAuth:"  + expectedUser);
        return true;
      }
    }
  }

  return false;
}


String decodeBase64(const String& input) {
  size_t outputLen = 0;
  size_t inputLen = input.length();
  unsigned char output[257];  // размер буфера, при необходимости увеличь

  int ret = mbedtls_base64_decode(output, sizeof(output), &outputLen,
                                  (const unsigned char *)input.c_str(), inputLen);

  if (ret != 0) {
    Serial.println("Base64 decode error");
    return "";
  }

  if (outputLen >= sizeof(output)) return "";
  output[outputLen] = '\0';
  return String((char *)output).substring(0, outputLen);
}

bool checkAuth(String &currentUser) {
  // Проверка логина и пароля администратора
  preferences.begin("auth", true);
  String adminLogin = preferences.getString("login", "admin");
  String adminPass = preferences.getString("password", "123456789");
  preferences.end();

  // Если логин/пароль совпали — считаем введённый логин текущим пользователем
  if (server.hasHeader("Authorization")) {
    String auth = server.header("Authorization");
    auth.replace("Basic ", "");
    auth.trim();
    String decoded = decodeBase64(auth);
    int separatorIndex = decoded.indexOf(':');
    if (separatorIndex != -1) {
      String inputLogin = decoded.substring(0, separatorIndex);
      String inputPass = decoded.substring(separatorIndex + 1);
      inputLogin.trim();
      inputPass.trim();

      if (inputLogin == adminLogin && inputPass == adminPass) {
        currentUser = inputLogin;
        return true;
      }
    }
  }

  // Если не админ – проверка мультипользователей
  if (multiUserAuth(currentUser)) {
    return true;
  }

  server.requestAuthentication();
  return false;
}

String formatTimeMinutes(int totalMinutes) {
  totalMinutes = ((totalMinutes % 1440) + 1440) % 1440;
  char buffer[6];
  snprintf(buffer, sizeof(buffer), "%02d:%02d", totalMinutes / 60, totalMinutes % 60);
  return String(buffer);
}

bool parseTimeMinutes(const String& value, int &minutes) {
  if (value.length() != 5 || value.charAt(2) != ':') return false;
  for (int i = 0; i < 5; i++) {
    if (i == 2) continue;
    if (!isDigit(value.charAt(i))) return false;
  }
  int hour = value.substring(0, 2).toInt();
  int minute = value.substring(3, 5).toInt();
  if (hour < 0 || hour > 23 || minute < 0 || minute > 59) return false;
  minutes = hour * 60 + minute;
  return true;
}

bool isAudioWorkTimeNow() {
  if (!audioWorkTimeEnabled) return true;

  struct tm timeinfo;
  if (!getLocalTime(&timeinfo, 0)) {
    Serial.println("[audio] work time enabled, but local time is not synced; playback blocked");
    return false;
  }

  int nowMinutes = timeinfo.tm_hour * 60 + timeinfo.tm_min;
  if (audioWorkStartMinute == audioWorkEndMinute) return true;
  if (audioWorkStartMinute < audioWorkEndMinute) {
    return nowMinutes >= audioWorkStartMinute && nowMinutes <= audioWorkEndMinute;
  }
  return nowMinutes >= audioWorkStartMinute || nowMinutes <= audioWorkEndMinute;
}

void stopAudioForWorkTime(const String& reason) {
  Serial.println("[audio] playback blocked by work time: " + reason);
  audio.stopSong();
  isConnected = false;
  isPaused = true;
  playlistActive = false;
  radioReconnectTimer = 0;
}

bool ensureAudioWorkTimeAllowed(bool sendResponse) {
  if (isAudioWorkTimeNow()) return true;
  stopAudioForWorkTime("outside allowed range " + formatTimeMinutes(audioWorkStartMinute) + "-" + formatTimeMinutes(audioWorkEndMinute));
  if (sendResponse) {
    server.send(403, "text/plain", "Playback blocked outside working time");
  }
  return false;
}

void enforceAudioWorkTime() {
  if (!audioWorkTimeEnabled) return;
  if (millis() - audioWorkTimeLastCheck < AUDIO_WORK_TIME_CHECK_MS) return;
  audioWorkTimeLastCheck = millis();
  if (!isPaused || isConnected || playlistActive) {
    ensureAudioWorkTimeAllowed(false);
  }
}

void handleAudioWorkTime() {
  String currentUser;
  if (!checkAuth(currentUser)) return;
  if (!isAdmin(currentUser)) {
    server.send(403, "text/plain", "Forbidden: Admin only");
    return;
  }

  bool newEnabled = !server.hasArg("enabled") || server.arg("enabled") != "0";
  int newStart = audioWorkStartMinute;
  int newEnd = audioWorkEndMinute;
  if (newEnabled) {
    if (!server.hasArg("start") || !server.hasArg("end")) {
      server.send(400, "text/plain", "Missing start/end parameters");
      return;
    }
    if (!parseTimeMinutes(server.arg("start"), newStart) || !parseTimeMinutes(server.arg("end"), newEnd)) {
      server.send(400, "text/plain", "Invalid time format, use HH:MM");
      return;
    }
  }

  audioWorkTimeEnabled = newEnabled;
  audioWorkStartMinute = newStart;
  audioWorkEndMinute = newEnd;
  preferences.begin("audio", false);
  preferences.putBool("workEnabled", audioWorkTimeEnabled);
  preferences.putInt("workStart", audioWorkStartMinute);
  preferences.putInt("workEnd", audioWorkEndMinute);
  preferences.end();
  ensureAudioWorkTimeAllowed(false);
  server.send(200, "text/plain", audioWorkTimeEnabled ? "OK" : "DISABLED");
}

String endpointNameFromUrl(const String& url) {
  int q = url.indexOf('?');
  String clean = (q >= 0) ? url.substring(0, q) : url;
  int slash = clean.lastIndexOf('/');
  if (slash >= 0 && slash + 1 < clean.length()) return clean.substring(slash + 1);
  return clean;
}

bool startPlaylistItem(int index) {
  if (index < 0 || index >= playlistCount) {
    return false;
  }

  playlistIndex = index;
  currentStreamUrl = playlistUrls[playlistIndex];
  radioUrl = currentStreamUrl.c_str();
  isCustomStation = true;
  customStationName = endpointNameFromUrl(currentStreamUrl);
  if (!ensureAudioWorkTimeAllowed(false)) return false;

  Serial.println("[audio] playlist item " + String(playlistIndex + 1) + "/" + String(playlistCount) + ": " + currentStreamUrl);
  audio.stopSong();
  delay(100);
  isConnected = connectToStation(customStationName.c_str(), radioUrl);
  isPaused = false;
  if (!isConnected) {
    Serial.println("[audio] playlist item connection failed, will skip after start timeout");
  }
  playlistSawRunning = false;
  playlistItemStartedAt = millis();
  playlistLastStateCheck = 0;
  playlistStoppedAt = 0;
  return isConnected;
}

void finishAudioPlaylist() {
  Serial.println("[audio] playlist finished");
  playlistActive = false;

  if (playlistReturnToPrevious && playlistPreviousStreamUrl.length() > 0) {
    currentStreamUrl = playlistPreviousStreamUrl;
    radioUrl = currentStreamUrl.c_str();
    isCustomStation = true;
    customStationName = endpointNameFromUrl(currentStreamUrl);
    Serial.println("[audio] returning to previous stream: " + currentStreamUrl);
    audio.stopSong();
    delay(100);
    if (!ensureAudioWorkTimeAllowed(false)) return;
    isConnected = connectToStation(customStationName.c_str(), radioUrl);
    isPaused = !isConnected;
  } else {
    audio.stopSong();
    isConnected = false;
    isPaused = true;
  }
}

void firmwareUpdateTask(void* parameter) {
  String firmwareUrl = *((String*)parameter);
  delete (String*)parameter;

  HTTPClient http;
  WiFiClient client;
  bool shouldRestart = false;

  setOtaState("OTA_START", 0);

  audio.stopSong();
  playlistActive = false;
  isConnected = false;
  isPaused = true;
  delay(200);

  if (!http.begin(client, firmwareUrl)) {
    setOtaState("OTA_ERROR_DOWNLOAD", 0);
    otaInProgress = false;
    vTaskDelete(NULL);
    return;
  }

  int httpCode = http.GET();
  if (httpCode != HTTP_CODE_OK) {
    Serial.println("[ota] HTTP error: " + String(httpCode));
    http.end();
    setOtaState("OTA_ERROR_DOWNLOAD", 0);
    otaInProgress = false;
    vTaskDelete(NULL);
    return;
  }

  int contentLength = http.getSize();
  if (contentLength <= 0) {
    Serial.println("[ota] Missing or invalid Content-Length");
    http.end();
    setOtaState("OTA_ERROR_DOWNLOAD", 0);
    otaInProgress = false;
    vTaskDelete(NULL);
    return;
  }

  uint32_t freeSketchSpace = ESP.getFreeSketchSpace();
  if ((uint32_t)contentLength > freeSketchSpace) {
    Serial.println("[ota] Not enough OTA space. Content-Length=" + String(contentLength) + ", free=" + String(freeSketchSpace));
    http.end();
    setOtaState("OTA_ERROR_SPACE", 0);
    otaInProgress = false;
    vTaskDelete(NULL);
    return;
  }

  if (!Update.begin(contentLength)) {
    Serial.println("[ota] Update.begin failed: " + String(Update.errorString()));
    http.end();
    setOtaState("OTA_ERROR_SPACE", 0);
    otaInProgress = false;
    vTaskDelete(NULL);
    return;
  }

  WiFiClient* stream = http.getStreamPtr();
  uint8_t buffer[1024];
  size_t writtenTotal = 0;
  int lastReportedProgress = 0;
  setOtaState("OTA_START", 0);

  while (http.connected() && writtenTotal < (size_t)contentLength) {
    size_t available = stream->available();
    if (available) {
      size_t toRead = min(available, sizeof(buffer));
      int readBytes = stream->readBytes(buffer, toRead);
      if (readBytes <= 0) {
        continue;
      }

      size_t written = Update.write(buffer, readBytes);
      if (written != (size_t)readBytes) {
        Serial.println("[ota] Update.write failed: " + String(Update.errorString()));
        Update.abort();
        http.end();
        setOtaState("OTA_ERROR_VERIFY", lastReportedProgress);
        otaInProgress = false;
        vTaskDelete(NULL);
        return;
      }

      writtenTotal += written;
      int progress = (writtenTotal * 100) / contentLength;
      int steppedProgress = (progress / OTA_PROGRESS_STEP) * OTA_PROGRESS_STEP;
      if (steppedProgress > lastReportedProgress) {
        lastReportedProgress = steppedProgress;
        setOtaState("OTA_" + String(steppedProgress), steppedProgress);
      }
    } else {
      delay(1);
    }
  }

  if (writtenTotal != (size_t)contentLength) {
    Serial.println("[ota] Download incomplete. Written=" + String(writtenTotal) + ", expected=" + String(contentLength));
    Update.abort();
    http.end();
    setOtaState("OTA_ERROR_DOWNLOAD", lastReportedProgress);
    otaInProgress = false;
    vTaskDelete(NULL);
    return;
  }

  if (!Update.end(true)) {
    Serial.println("[ota] Update.end failed: " + String(Update.errorString()));
    http.end();
    setOtaState("OTA_ERROR_VERIFY", lastReportedProgress);
    otaInProgress = false;
    vTaskDelete(NULL);
    return;
  }

  http.end();
  setOtaState("OTA_OK", 100);
  shouldRestart = true;
  otaInProgress = false;

  if (shouldRestart) {
    delay(1500);
    ESP.restart();
  }

  vTaskDelete(NULL);
}

void handleFirmwareUpdate() {
  String currentUser;
  if (!checkAuth(currentUser)) return;
  if (!isAdmin(currentUser)) {
    server.send(403, "text/plain", "Forbidden: Admin only");
    return;
  }

  if (otaInProgress) {
    server.send(409, "text/plain", "OTA already in progress");
    return;
  }

  String firmwareUrl = server.arg("url");
  if (firmwareUrl.length() == 0) {
    String body = server.arg("plain");
    body.trim();
    if (body.startsWith("UPDATE|")) {
      firmwareUrl = body.substring(7);
    } else if (body.startsWith("{")) {
      DynamicJsonDocument doc(512);
      DeserializationError error = deserializeJson(doc, body);
      if (!error) {
        firmwareUrl = doc["url"].as<String>();
      }
    } else if (body.startsWith("http://")) {
      firmwareUrl = body;
    }
  }

  firmwareUrl.trim();
  if (!firmwareUrl.startsWith("http://")) {
    server.send(400, "text/plain", "Invalid firmware URL. Only http:// is supported");
    return;
  }

  otaInProgress = true;
  setOtaState("OTA_START", 0);
  String* taskUrl = new String(firmwareUrl);
  BaseType_t created = xTaskCreatePinnedToCore(
    firmwareUpdateTask,
    "firmware_ota",
    8192,
    taskUrl,
    1,
    NULL,
    1
  );

  if (created != pdPASS) {
    delete taskUrl;
    otaInProgress = false;
    setOtaState("OTA_ERROR_DOWNLOAD", 0);
    server.send(500, "text/plain", "OTA task failed");
    return;
  }

  server.send(202, "text/plain", "OTA_START");
}

void playNextPlaylistItem() {
  if (!playlistActive) return;

  int nextIndex = playlistIndex + 1;
  if (!ensureAudioWorkTimeAllowed(false)) return;

  if (nextIndex < playlistCount) {
    startPlaylistItem(nextIndex);
  } else {
    finishAudioPlaylist();
  }
}

void handleAudioPlaylist() {
  String currentUser;
  if (!checkAuth(currentUser)) return;
  if (!isAdmin(currentUser)) {
    server.send(403, "text/plain", "Forbidden: Admin only");
    return;
  }

  String body = server.arg("plain");
  if (body.length() == 0) {
    server.send(400, "text/plain", "Missing JSON body");
    return;
  }

  DynamicJsonDocument doc(2048);
  DeserializationError error = deserializeJson(doc, body);
  if (error) {
    server.send(400, "text/plain", "Invalid JSON");
    return;
  }

  JsonArray urls = doc["urls"].as<JsonArray>();
  if (urls.isNull() || urls.size() == 0) {
    server.send(400, "text/plain", "Missing urls array");
    return;
  }
  if (urls.size() > AUDIO_PLAYLIST_MAX_ITEMS) {
    server.send(400, "text/plain", "Too many urls");
    return;
  }

  if (!ensureAudioWorkTimeAllowed(true)) return;

  resetAudioPlaylist();
  playlistPreviousStreamUrl = currentStreamUrl;
  playlistReturnToPrevious = doc["returnToPrevious"] | true;

  for (JsonVariant value : urls) {
    String url = value.as<String>();
    url.trim();
    if (!url.startsWith("http://") && !url.startsWith("https://")) {
      resetAudioPlaylist();

      return;
    }

  }

  playlistActive = true;
  bool started = startPlaylistItem(0);
  server.send(started ? 200 : 500, "text/plain", started ? "PLAYLIST_STARTED" : "PLAYLIST_FAILED");
}

void handleAudioPlay() {
  Serial.println("[audio] /audio/play requested");
  String currentUser;
  if (!checkAuth(currentUser)) {
    Serial.println("[audio] /audio/play auth failed");
    return;
  }

  if (currentStreamUrl.length() > 0) {
    radioUrl = currentStreamUrl.c_str();
  }

  if (!ensureAudioWorkTimeAllowed(true)) return;

  if (radioUrl == nullptr || String(radioUrl).length() == 0) {
    Serial.println("[audio] play failed: missing URL");
    server.send(400, "text/plain", "Missing audio URL");
    return;
  }

  Serial.println("[audio] play before: isConnected=" + String(isConnected) + ", isPaused=" + String(isPaused) + ", url=" + String(radioUrl));
  if (isPaused || !isConnected) {
    const char* stationName = isCustomStation ? customStationName.c_str() : radioStations[currentStationIndex].name;
    audio.stopSong();
    delay(100);
    isConnected = connectToStation(stationName, radioUrl);
    isPaused = !isConnected;
  } else {
    isPaused = false;
  }

  Serial.println("[audio] play after: isConnected=" + String(isConnected) + ", isPaused=" + String(isPaused));
  server.send(200, "text/plain", isPaused ? "PAUSED" : "PLAYING");
}

  isConnected = false;

}

void handleSetUserAccess() {
  // Только админ может изменять настройки пользователей
  String currentUser;
  if (!checkAuth(currentUser)) return;
  if (!isAdmin(currentUser)) {
    server.send(403, "text/plain", "Only admin can change user settings");
    return;
  }

  preferences.begin("users", false);
  for (int i = 1; i <= 4; i++) {
    String uname = "user" + String(i);

    // --- Сохраняем пароль только если он непустой ---
    if (server.hasArg(uname + "_pass")) {
      String newPass = server.arg(uname + "_pass");
      if (newPass.length() > 0) {
        String checkPass = preferences.getString((uname + "_pass").c_str(), "");
        Serial.println("Check read before: " + uname + "_pass = " + checkPass);
        preferences.putString((uname + "_pass").c_str(), newPass);
        Serial.println("Updated " + uname + "_pass = " + newPass);
      } else {
        Serial.println("Skipped empty password for " + uname);
      }
    }

    // --- Чекбоксы: сбрасываем все флаги и включаем только те, что отмечены ---
    for (int j = 1; j <= 4; j++) {
      String key = uname + "_r" + String(j);
      bool isChecked = server.hasArg(key);  // передан ли чекбокс

      preferences.putBool(key.c_str(), isChecked);  // сохраняем true или false
      Serial.println("Set " + key + " = " + String(isChecked ? "YES" : "NO"));
    }
  }
  preferences.end();

  server.send(200, "text/plain", "User access settings updated");
}

  String currentUser;
  // Проверяем общую аутентификацию: если админ – имеет полный доступ, иначе – проверяем права
  if (!checkAuth(currentUser)) return;

  Serial.println("Converted ID: " + String(id));

  Serial.println("checkUserAccess00_id: " + id);
  Serial.println("handleToggle_relay: " + server.arg("relay"));
  if (id < 0 || id >= RELAY_COUNT) {
    server.send(400, "text/plain", "Invalid relay ID");
    return;
  }
  
  

  // Если не админ, проверяем наличие права управления этим реле
  //Serial.println("checkUserAccess0_currentUser: " + currentUser);
  //Serial.println("checkUserAccess0_id: " + String(id));

  Serial.println("User: " + currentUser + ", Relay ID: " + String(id));
  Serial.println("Access allowed: " + String(access ? "YES" : "NO"));
  if (!isAdmin(currentUser) && !checkUserAccess(currentUser, id)) {
    Serial.println("checkUserAccess_currentUser: " + currentUser);
    Serial.println("checkUserAccess_id: " + id);
    server.send(403, "text/plain", "Access Denied");
    return;
  }
  
  // Далее переключаем реле как раньше
  if (useTimer[id]) {
    if (relayTimers[id].active()) relayTimers[id].detach();
    if (!relayStates[id]) {
      relayTimers[id].once(timerSeconds[id], [id]() {
        relayStates[id] = true;
        digitalWrite(RELAY_PINS[id], HIGH);
        preferences.begin("relay", false);
        preferences.putBool(("relay" + String(id)).c_str(), true);
        preferences.end();
      });
    } else {
      relayTimers[id].once(timerSeconds[id], [id]() {
        relayStates[id] = false;
        digitalWrite(RELAY_PINS[id], LOW);
        preferences.begin("relay", false);
        preferences.putBool(("relay" + String(id)).c_str(), false);
        preferences.end();
      });
    }
  } else {
    relayStates[id] = !relayStates[id];
    digitalWrite(RELAY_PINS[id], relayStates[id] ? HIGH : LOW);
    preferences.begin("relay", false);
    preferences.putBool(("relay" + String(id)).c_str(), relayStates[id]);
    preferences.end();
  }
  server.send(200, "text/plain", "Toggled");
}

void handleTimer() {
  String currentUser;
  if (!checkAuth(currentUser)) return;
    
  int id = server.arg("id").toInt();
  Serial.println("handleTimer_id: " + id );
  if (id < 0 || id >= RELAY_COUNT) {
    server.send(400, "text/plain", "Invalid relay ID");
    return;
  }
  // Если не админ – проверяем право на это реле
  if (!isAdmin(currentUser) && !checkUserAccess(currentUser, id)) {
    server.send(403, "text/plain", "Access Denied");
    return;
  }
    
  useTimer[id] = server.arg("use") == "1";
  timerSeconds[id] = server.arg("sec").toInt();
  timerCycleMode[id] = server.hasArg("cycle") && server.arg("cycle") == "1";
  if (useTimer[id] && timerSeconds[id] == 0) {
    server.send(400, "text/plain", "Error: Timer duration cannot be zero.");
    return;
  }
  
  // Сохраняем состояние таймера для реле id (функция saveState() без изменений)
  preferences.begin("relay", false);
  preferences.putBool(("relay" + String(id)).c_str(), relayStates[id]);
  preferences.putBool(("useTimer" + String(id)).c_str(), useTimer[id]);
  preferences.putInt(("timerSec" + String(id)).c_str(), timerSeconds[id]);
  preferences.putString(("name" + String(id)).c_str(), relayNames[id]);
  preferences.putBool(("cycleMode" + String(id)).c_str(), timerCycleMode[id]);
  preferences.end();
  
  if (relayTimers[id].active()) relayTimers[id].detach();
  
  if (useTimer[id]) {
    if (timerCycleMode[id]) {
      relayTimers[id].attach(timerSeconds[id], [id]() {
        relayStates[id] = !relayStates[id];
        digitalWrite(RELAY_PINS[id], relayStates[id] ? HIGH : LOW);
        preferences.begin("relay", false);
        preferences.putBool(("relay" + String(id)).c_str(), relayStates[id]);
        preferences.end();
      });
    } else {
      if (!relayStates[id]) {
        relayTimers[id].once(timerSeconds[id], [id]() {
          relayStates[id] = true;
          digitalWrite(RELAY_PINS[id], HIGH);
          preferences.begin("relay", false);
          preferences.putBool(("relay" + String(id)).c_str(), true);
          preferences.end();
        });
      } else {
        relayTimers[id].once(timerSeconds[id], [id]() {
          relayStates[id] = false;
          digitalWrite(RELAY_PINS[id], LOW);
          preferences.begin("relay", false);
          preferences.putBool(("relay" + String(id)).c_str(), false);
          preferences.end();
        });
      }
    }
  }
  server.send(200, "text/plain", "OK");
}

void setup() {
  Serial.begin(115200);
  WiFiManager wifiManager;
  wifiManager.addParameter(&custom_login);
  wifiManager.addParameter(&custom_password);
  wifiManager.setTimeout(180);
  wifiManager.setAPCallback(configModeCallback);
  WiFi.setHostname(hostname);

  uint8_t mac[6];
  WiFi.macAddress(mac);
  char suffix[7];
  sprintf(suffix, "%02X%02X%02X", mac[3], mac[4], mac[5]);
  deviceName = "audio-" + String(suffix);
  Serial.print("Device Name: ");
  Serial.println(deviceName);

  if (!wifiManager.autoConnect("AUDIO CONTROL", "12345678")) {
    Serial.println("Can't connect! Rebooting...");
    delay(3000);
    ESP.restart();    
  }
  Serial.println("WiFi connected: " + WiFi.localIP().toString());
  udp.begin(localUdpPort);

  // Load saved NTP server and start client after WiFi is up
  preferences.begin("ntp", true);
  ntpServer = preferences.getString("server", "");
  preferences.end();
  startNtpClient();

  if (shouldSaveConfig) {
    preferences.begin("auth", false);
    String newLogin = custom_login.getValue();
    String newPassword = custom_password.getValue();
    if (newLogin.length() > 0 && newPassword.length() > 0) {
      preferences.putString("login", newLogin);
      preferences.putString("password", newPassword);
      Serial.println("Saved admin login/password from WiFiManager setup mode.");
    }
    preferences.end();
  }  

  //for (int i = 0; i < RELAY_COUNT; i++) {
    //pinMode(RELAY_PINS[i], OUTPUT);
  //}

  audio.setPinout(I2S_BCLK, I2S_LRC, I2S_DOUT);
  audio.setVolume(currentVolume);
  audio.forceMono(false);
  audio.setConnectionTimeout(10000, 10000);
  preferences.begin("audio", true);
  currentStreamUrl = preferences.getString("streamUrl", radioStations[currentStationIndex].url);
  preferences.end();
  loadAudioWorkTime();
  radioUrl = currentStreamUrl.c_str();
  customStationName = endpointNameFromUrl(currentStreamUrl);
  isCustomStation = true;
  isConnected = connectToStation(customStationName.c_str(), radioUrl);
  isPaused = !isConnected;

  preferences.begin("relay", false);
  for (int i = 0; i < RELAY_COUNT; i++) {
    relayStates[i] = preferences.getBool(("relay" + String(i)).c_str(), false);
    useTimer[i] = preferences.getBool(("useTimer" + String(i)).c_str(), false);
    timerCycleMode[i] = preferences.getBool(("cycleMode" + String(i)).c_str(), false);
    timerSeconds[i] = preferences.getInt(("timerSec" + String(i)).c_str(), 5);
    relayNames[i] = preferences.getString(("name" + String(i)).c_str(), "Relay " + String(i + 1));
    digitalWrite(RELAY_PINS[i], relayStates[i]);

    if (relayStates[i] && useTimer[i]) {
      relayTimers[i].once(timerSeconds[i], [i]() {
        relayStates[i] = false;
        digitalWrite(RELAY_PINS[i], LOW);
        preferences.begin("relay", false);
        preferences.putBool(("relay" + String(i)).c_str(), false);
        preferences.end();
      });
    }
  }
  locationName = preferences.getString("location", "");
  preferences.end();

  // Зарегистрируем обработчики
  server.on("/status", []() {
     DynamicJsonDocument doc(2048);

    JsonArray relays = doc.createNestedArray("relays");
    for (int i = 0; i < RELAY_COUNT; i++) {      
      JsonObject relay = relays.createNestedObject();
      relay["id"] = i;
      relay["state"] = relayStates[i];
      relay["useTimer"] = useTimer[i];
      relay["seconds"] = timerSeconds[i];
      relay["name"] = relayNames[i];
    }

    JsonObject wifi = doc.createNestedObject("wifi");
    wifi["rssi"] = WiFi.RSSI();
    wifi["ssid"] = WiFi.SSID();

    JsonObject audioStatus = doc.createNestedObject("audio");
    audioStatus["state"] = (!isPaused && isConnected) ? "play" : "pause";
    audioStatus["current"] = endpointNameFromUrl(currentStreamUrl);
    audioStatus["url"] = currentStreamUrl;
    audioStatus["volume"] = currentVolume;
    audioStatus["playlistActive"] = playlistActive;
    audioStatus["workTimeEnabled"] = audioWorkTimeEnabled;
    audioStatus["workStart"] = formatTimeMinutes(audioWorkStartMinute);
    audioStatus["workEnd"] = formatTimeMinutes(audioWorkEndMinute);
    audioStatus["workAllowed"] = isAudioWorkTimeNow();
    if (playlistActive) {
      audioStatus["playlistIndex"] = playlistIndex;
      audioStatus["playlistCount"] = playlistCount;
    }

    JsonObject firmwareStatus = doc.createNestedObject("firmware");
    firmwareStatus["version"] = FW_VERSION;
    firmwareStatus["status"] = otaStatus;
    firmwareStatus["progress"] = otaProgress;
    firmwareStatus["updating"] = otaInProgress;

    JsonObject ntpStatus = doc.createNestedObject("ntp");
    ntpStatus["server"] = ntpServer;

    struct tm timeinfo;
    char timeStr[32] = "";
    if (getLocalTime(&timeinfo, 0)) {
      strftime(timeStr, sizeof(timeStr), "%Y-%m-%d %H:%M:%S", &timeinfo);
      ntpStatus["time"] = timeStr;
      ntpStatus["synced"] = true;
    } else {

    }

  });

void loop() {
  server.handleClient();
  
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi not connected!");
    return;
  }
  enforceAudioWorkTime();
  if (isConnected && isAudioWorkTimeNow()) {
    audio.loop();
  }  
  // Автоматическое восстановление интернет-радио
  if (!playlistActive &&
      !isPaused &&
      WiFi.status() == WL_CONNECTED &&
      isAudioWorkTimeNow() &&
      !audio.isRunning()) {

      if (radioReconnectTimer == 0) {
          radioReconnectTimer = millis();
      }

      if (millis() - radioReconnectTimer >= 5000) {

          Serial.println("[audio] Stream lost, reconnecting...");

          audio.stopSong();
          delay(100);

          isConnected = connectToStation(
              customStationName.c_str(),
              radioUrl
          );

          radioReconnectTimer = 0;
      }
  } else {
      radioReconnectTimer = 0;
  }
    
  if (playlistActive && isConnected && !isPaused && millis() - playlistLastStateCheck >= AUDIO_PLAYLIST_STATE_CHECK_MS) {
    playlistLastStateCheck = millis();

    bool running = audio.isRunning();

    if (running) {
      playlistSawRunning = true;
      playlistStoppedAt = 0;
    }

    if (playlistSawRunning && !running && millis() - playlistItemStartedAt > 1500) {
      unsigned long itemAge = millis() - playlistItemStartedAt;
      unsigned long stopGrace = itemAge < AUDIO_PLAYLIST_EARLY_END_PROTECT_MS
        ? AUDIO_PLAYLIST_EARLY_END_GRACE_MS
        : AUDIO_PLAYLIST_END_GRACE_MS;

      if (playlistStoppedAt == 0) {
        playlistStoppedAt = millis();
        Serial.println("[audio] playlist item looks stopped, waiting for EOF grace period");
      } else if (millis() - playlistStoppedAt >= stopGrace) {
        Serial.println("[audio] playlist item finished by isRunning() fallback");
        isConnected = false;
        playNextPlaylistItem();
      }
    }    
  }

  bool running = audio.isRunning();

  if (running) {
    playlistSawRunning = true;
    playlistStoppedAt = 0;
  }
  
  
  int packetSize = udp.parsePacket();
  if (packetSize) {
    int len = udp.read(incomingPacket, 255);
    if (len > 0) {
      incomingPacket[len] = '\0';
    }
    Serial.printf("Received UDP packet: '%s' from %s:%d\n", incomingPacket,
                  udp.remoteIP().toString().c_str(), udp.remotePort());
    if (strcmp(incomingPacket, "ping") == 0) {
      const char* host = WiFi.getHostname();
      String safeHostname = (host != nullptr) ? String(host) : "unknown";
      String safeName = deviceName.length() > 0 ? deviceName : "ESP32";
      String mac = WiFi.macAddress();
      String info = "ESP32-POWERCTRL|MAC=" + mac +
                    "|NAME=" + locationName +
                    "|HOSTNAME=" + safeHostname;
      udp.beginPacket(udp.remoteIP(), udp.remotePort());
      udp.write((const uint8_t*)info.c_str(), info.length());
      udp.endPacket();
      Serial.println("Sent response: " + info);
    }
  }
  unsigned long itemAge = millis() - playlistItemStartedAt;
  if (!playlistSawRunning && itemAge >= AUDIO_PLAYLIST_START_TIMEOUT_MS) {
    Serial.println("[audio] playlist item did not start, skipping");
    isConnected = false;
    playNextPlaylistItem();
  }
  delay(10);
}
